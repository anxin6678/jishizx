// === 新增逻辑：获取或设置固定 ID === // 1. 尝试从浏览器的本地存储中读取之前保存的 ID let mySavedId = localStorage.getItem('my_fixed_peer_id'); // 2. 如果没有找到（说明是第一次打开网页） if (!mySavedId) { // 弹窗让用户自己设定一个 ID mySavedId = prompt("欢迎来到语音聊天室！\n\n请为你自己设定一个【固定的通讯 ID】\n（建议使用英文、拼音或数字，例如：zhangsan88）:"); // 如果用户点了取消或者什么都没填，就给他生成一个随机的作为备用 if (!mySavedId || mySavedId.trim() === '') { mySavedId = 'user_' + Math.random().toString(36).substr(2, 6); alert("你没有输入自定义 ID，系统已为你随机分配：" + mySavedId); } // 3. 将这个 ID 永久保存在浏览器中 localStorage.setItem('my_fixed_peer_id', mySavedId.trim()); } // =================================== // 使用固定的 ID 初始化 PeerJS const peer = new Peer(mySavedId); let myId = ''; let localStream = null; // 存储连接对象，支持多人 const dataConns = {}; const mediaCalls = {}; // DOM 元素 const chatBox = document.getElementById('chat-box'); const activePeersDiv = document.getElementById('active-peers'); const audioContainer = document.getElementById('audio-container'); // 1. 成功连接服务器，获取自身 ID peer.on('open', (id) => { myId = id; document.getElementById('my-id').innerText = id; logSystem('已连接网络！你的固定 ID 是: ' + id); }); // 监听如果 ID 被别人占用了的错误 peer.on('error', (err) => { if (err.type === 'unavailable-id') { alert('糟糕，这个 ID [' + mySavedId + '] 当前正被别人使用！\n页面将刷新，请重新设置一个新的 ID。'); // 清除旧的冲突 ID，刷新页面让用户重新输入 localStorage.removeItem('my_fixed_peer_id'); location.reload(); } else { logSystem('网络错误: ' + err.type); console.error(err); } }); // 2. 开启麦克风 document.getElementById('btn-mic').addEventListener('click', async () => { try { localStream = await navigator.mediaDevices.getUserMedia({ audio: true, video: false }); const btn = document.getElementById('btn-mic'); btn.className = 'success'; btn.innerText = '麦克风已开启 ✅'; logSystem('麦克风已就绪。'); } catch (err) { alert('无法访问麦克风，请检查浏览器权限限制。'); } }); // 3. 接收他人的文字连接请求 peer.on('connection', (conn) => { setupDataConnection(conn); }); // 4. 接收他人的语音呼叫 peer.on('call', (call) => { if (!localStream) { logSystem('收到语音呼叫，但你未开启麦克风，无法接听。'); return; } call.answer(localStream); setupMediaCall(call); }); // 5. 主动连接其他用户 (文字+语音) function connectToPeer() { const targetId = document.getElementById('target-id').value.trim(); if (!targetId || targetId === myId) return alert('请输入有效的他人 ID'); if (!localStream) return alert('请先开启麦克风再进行连接！'); const conn = peer.connect(targetId); setupDataConnection(conn); const call = peer.call(targetId, localStream); setupMediaCall(call); document.getElementById('target-id').value = ''; } // ================= 工具函数 ================= function setupDataConnection(conn) { conn.on('open', () => { dataConns[conn.peer] = conn; updatePeerList(); logSystem(`已与用户 [${conn.peer}] 建立连接`); }); conn.on('data', (data) => { logMessage(`[${conn.peer}]: ${data}`, 'peer'); }); conn.on('close', () => { delete dataConns[conn.peer]; updatePeerList(); logSystem(`用户 [${conn.peer}] 断开连接`); }); } function setupMediaCall(call) { call.on('stream', (remoteStream) => { if (document.getElementById('audio-' + call.peer)) return; const audioElement = document.createElement('audio'); audioElement.id = 'audio-' + call.peer; audioElement.srcObject = remoteStream; audioElement.autoplay = true; audioContainer.appendChild(audioElement); }); call.on('close', () => { const audioEl = document.getElementById('audio-' + call.peer); if (audioEl) audioEl.remove(); }); mediaCalls[call.peer] = call; } function sendMsg() { const input = document.getElementById('msg-input'); const text = input.value.trim(); if (!text) return; logMessage(`[我]: ${text}`, 'me'); for (let peerId in dataConns) { dataConns[peerId].send(text); } input.value = ''; } function updatePeerList() { const ids = Object.keys(dataConns); activePeersDiv.innerText = ids.length ? `当前连接: ${ids.join(', ')}` : '当前连接: 无'; } function logMessage(msg, type) { const div = document.createElement('div'); div.className = `msg ${type}`; div.innerText = msg; chatBox.appendChild(div); chatBox.scrollTop = chatBox.scrollHeight; } function logSystem(msg) { logMessage(`— ${msg} —`, 'system'); }
